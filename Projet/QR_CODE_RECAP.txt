🎯 UTILISATION SIMPLE
════════════════════════════════════════════════════════════════════════════════════

Import:
  import org.example.service.QRService;

Générateur:
  QRService qrService = QRService.getInstance();

3 méthodes essentielles:

  1) generateServiceQRCode(id, "data")
     → Simple: "Service#123"

  2) generateServiceQRCodeWithInfo(id, nom, type, specialite)
     → Structuré: "id:123|nom:...|type:...|specialite:..."

  3) generateServiceQRCodeWithLink(id, "http://localhost")
     → URL: "http://localhost/service/123"

Affichage:
  Image img = qrService.loadQRCodeImage(filePath);
  imageView.setImage(img);

💻 INTÉGRATION DANS VOTRE CODE
════════════════════════════════════════════════════════════════════════════════════

EXEMPLE 1: Afficher QR d'un service
────────────────────────────────────
@FXML private ImageView qrView;

QRService qr = QRService.getInstance();
String path = qr.generateServiceQRCodeWithInfo(
    service.getId(), service.getNom(),
    service.getType(), service.getSpecialite()
);

Image img = qr.loadQRCodeImage(path);
qrView.setImage(img);


EXEMPLE 2: Générer pour tous les services
──────────────────────────────────────────
new Thread(() -> {
    QRService qr = QRService.getInstance();
    ServiceService ss = ServiceService.getInstance();

    for (Service s : ss.getAll()) {
        qr.generateServiceQRCodeWithInfo(
            s.getId(), s.getNom(), s.getType(), s.getSpecialite()
        );
    }
}).start();


EXEMPLE 3: Générer automatiquement (dans ServiceService.add)
──────────────────────────────────────────────────────────────
if (rs.next()) {
    service.setId(rs.getInt(1));

    QRService qr = QRService.getInstance();
    qr.generateServiceQRCodeWithInfo(
        service.getId(), service.getNom(),
        service.getType(), service.getSpecialite()
    );
}
