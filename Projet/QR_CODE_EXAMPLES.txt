// Dans votre contrôleur
@FXML private ImageView serviceQRView;
@FXML private Label serviceInfoLabel;

public void onServiceSelected(Service service) {
    QRService qrService = QRService.getInstance();

    // Générer le QR code
    String filePath = qrService.generateServiceQRCodeWithInfo(
        service.getId(),
        service.getNom(),
        service.getType(),
        service.getSpecialite()
    );

    // Afficher l'image
    if (filePath != null) {
        Image qrImage = qrService.loadQRCodeImage(filePath);
        serviceQRView.setImage(qrImage);
        serviceInfoLabel.setText("✅ QR code généré pour: " + service.getNom());
    }
}

// Dans votre classe principale ou contrôleur principal
public void initializeQRCodes() {
    // Exécuter dans un thread pour ne pas bloquer l'UI
    new Thread(() -> {
        QRService qrService = QRService.getInstance();
        ServiceService serviceService = ServiceService.getInstance();

        List<Service> allServices = serviceService.getAll();
        System.out.println("🔄 Génération de " + allServices.size() + " QR codes...");

        for (Service service : allServices) {
            qrService.generateServiceQRCodeWithInfo(
                service.getId(),
                service.getNom(),
                service.getType(),
                service.getSpecialite()
            );
        }

        System.out.println("✅ Tous les QR codes sont prêts!");
    }).start();
}

// MODIFIER ServiceService.java - Méthode add()

public boolean add(Service service) throws ValidationException {
    if (!validateService(service)) {
        throw new ValidationException("Validation échouée: " + validationErrors);
    }

    String sql = "INSERT INTO service (...) VALUES (...)";
    try (Connection conn = DatabaseUtil.getConnection();
         PreparedStatement stmt = conn.prepareStatement(sql,
                                   Statement.RETURN_GENERATED_KEYS)) {

        // ... code SQL existant ...

        try (ResultSet rs = stmt.getGeneratedKeys()) {
            if (rs.next()) {
                service.setId(rs.getInt(1));

                // 🆕 GÉNÉRER LE QR CODE AUTOMATIQUEMENT
                QRService qrService = QRService.getInstance();
                String qrPath = qrService.generateServiceQRCodeWithInfo(
                    service.getId(),
                    service.getNom(),
                    service.getType(),
                    service.getSpecialite()
                );

                if (qrPath != null) {
                    System.out.println("✨ QR code généré: " + qrPath);
                } else {
                    System.err.println("⚠️ QR code non généré (mais service créé)");
                }
            }
        }
        return true;
    } catch (SQLException e) {
        // ... error handling ...
    }
}

// Créer une colonne ImageView pour afficher les QR codes

TableColumn<Service, Image> qrColumn = new TableColumn<>("QR Code");
qrColumn.setPrefWidth(150);

qrColumn.setCellValueFactory(cellData -> {
    Service service = cellData.getValue();
    QRService qrService = QRService.getInstance();

    String filePath = "qr_codes/service_" + service.getId() + ".png";
    Image qrImage = qrService.loadQRCodeImage(filePath);

    return new javafx.beans.property.SimpleObjectProperty<>(qrImage);
});

qrColumn.setCellFactory(column -> new TableCell<Service, Image>() {
    private final ImageView imageView = new ImageView();

    {
        imageView.setFitWidth(100);
        imageView.setFitHeight(100);
        imageView.setPreserveRatio(true);
    }

    @Override
    protected void updateItem(Image image, boolean empty) {
        super.updateItem(image, empty);
        if (empty || image == null) {
            setGraphic(null);
        } else {
            imageView.setImage(image);
            setGraphic(imageView);
        }
    }
});

serviceTable.getColumns().add(qrColumn);

public void deleteServiceWithQR(int serviceId) {
    QRService qrService = QRService.getInstance();
    ServiceService serviceService = ServiceService.getInstance();

    // Supprimer le QR code
    boolean qrDeleted = qrService.deleteServiceQRCode(serviceId);

    // Supprimer le service
    serviceService.delete(serviceId);

    if (qrDeleted) {
        System.out.println("✅ Service et QR code supprimés");
    }
}
